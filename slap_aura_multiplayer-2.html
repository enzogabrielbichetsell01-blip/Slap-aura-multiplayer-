// SLAP AURA - servidor multiplayer Node.js
// Instalação: npm install
// Iniciar: npm start
const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: '*' } });
const PORT = process.env.PORT || 3000;
const gloves = [{cost:0,power:1},{cost:10,power:2},{cost:35,power:4},{cost:100,power:8},{cost:250,power:15},{cost:600,power:30}];
const players = {};
const clamp = (n,min,max)=>Math.max(min,Math.min(max,Number(n)||0));
function publicState(){return {players};}
io.on('connection', socket => {
socket.on('join', data => {
players[socket.id] = {id:socket.id,name:String(data?.name||'Jogador').slice(0,16),x:0,z:0,aura:0,power:1,glove:0,hits:0,face:'😎'};
io.emit('state', publicState());
});
socket.on('move', data => {
const p=players[socket.id]; if(!p)return;
p.x=clamp(data?.x,-9,9); p.z=clamp(data?.z,-5,9);
io.emit('state',publicState());
});
socket.on('buyGlove', data => {
const p=players[socket.id]; if(!p)return;
const i=Math.floor(Number(data?.index)); const g=gloves[i];
if(!g || i<=p.glove || p.aura<g.cost)return;
p.aura-=g.cost;p.glove=i;p.power=g.power;
io.emit('state',publicState());
});
socket.on('slap', () => {
const a=players[socket.id]; if(!a)return;
let target=null,dist=2.35;
for(const b of Object.values(players)){
if(b.id===a.id)continue;
const d=Math.hypot(a.x-b.x,a.z-b.z);
if(d<dist){dist=d;target=b;}
}
if(!target){socket.emit('slapResult',{ok:false,attacker:socket.id,message:'Chegue mais perto de outro jogador!'});return;}
target.face='😵';a.aura+=a.power;a.hits++;
io.emit('slapResult',{ok:true,attacker:a.id,target:target.id,gained:a.power});
io.emit('state',publicState());
setTimeout(()=>{if(players[target.id]){players[target.id].face='😲';io.emit('state',publicState());}},700);
});
socket.on('disconnect',()=>{delete players[socket.id];io.emit('state',publicState());});
});
app.get('/',(req,res)=>res.send('SLAP AURA multiplayer server está online!'));
server.listen(PORT,()=>console.log('SLAP AURA server rodando na porta '+PORT));
